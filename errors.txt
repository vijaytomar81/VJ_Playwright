# ============================================================
# ALLIANZ PLAYWRIGHT FRAMEWORK - WINDOWS VALIDATION
# ============================================================

# 1. Enable Windows trusted certificates for Node.js
$env:NODE_USE_SYSTEM_CA = "1"

# 2. Verify installed versions
node --version
npm --version
npx --no-install playwright --version
npx --no-install tsc --version

# 3. Validate TypeScript
npx --no-install tsc --noEmit
if ($LASTEXITCODE -ne 0) { throw "TypeScript validation failed." }

# 4. Install Playwright Chromium browser
npx --no-install playwright install chromium
if ($LASTEXITCODE -ne 0) { throw "Chromium installation failed." }

# 5. Initialize EvidenceFactory run
npx --no-install tsx tests/framework/evidence-factory/initialize-run.ts
if ($LASTEXITCODE -ne 0) { throw "EvidenceFactory initialization failed." }

# 6. Execute EvidenceFactory scenarios
npx --no-install playwright test tests/framework/evidence-factory/evidence-factory-excel.spec.ts --project=framework --workers=4
if ($LASTEXITCODE -ne 0) { throw "EvidenceFactory scenario tests failed." }

# 7. Generate and validate Excel evidence report
npx --no-install playwright test tests/framework/evidence-factory/evidence-report.spec.ts --project=framework --workers=1
if ($LASTEXITCODE -ne 0) { throw "Excel report generation test failed." }

# 8. Run POM Generator tests
npx --no-install playwright test tests/framework/pom-generator --project=framework --workers=4
if ($LASTEXITCODE -ne 0) { throw "POM Generator tests failed." }

# 9. Read current run ID
$runId = (Get-Content .\results\.current-test-run.json | ConvertFrom-Json).runId

# 10. Open generated Excel report folder
$reportDir = ".\results\$runId\report"

if (Test-Path $reportDir) {
    explorer $reportDir
} else {
    Write-Host "Report folder not found: $reportDir"
}

# 11. Display completion message
Write-Host ""
Write-Host "=========================================="
Write-Host " PLAYWRIGHT FRAMEWORK VALIDATION COMPLETE"
Write-Host "=========================================="
Write-Host "Run ID: $runId"
Write-Host "Report Directory: $reportDir"

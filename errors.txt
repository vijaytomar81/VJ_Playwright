import { test, type Page, type TestInfo } from '@playwright/test';
import { platform, release } from 'node:os';

const NOT_KNOWN = '';

/**
 * Captures runtime browser metadata from the existing
 * Playwright Page and stores it in test annotations.
 *
 * - Does not launch an additional browser.
 * - Does not execute browser binaries.
 * - Supports Chromium, Firefox and WebKit.
 * - Supports desktop and mobile-emulated contexts.
 * - Captures the Host OS and optional Remote Device information.
 * - Avoids duplicate metadata annotations.
 * - Safely handles calls outside an active Playwright test.
 */
export function captureExecutionEnvironmentMetadata(page: Page): void {
  let testInfo: TestInfo;

  /*
   * Page objects may be initialized outside an active
   * Playwright test. Metadata collection must not prevent
   * normal page object initialization.
   */
  try {
    testInfo = test.info();
  } catch {
    return;
  }

  const browser = page.context().browser();
  const viewport = page.viewportSize();

  /*
   * Capture the operating system running the Playwright
   * test process, including its runtime OS version.
   */
  const operatingSystems: Record<string, string> = {
    win32: 'Windows',
    linux: 'Linux',
    darwin: 'macOS',
  };

  const hostPlatform = platform();
  const hostOSVersion = release();
  const hostOS = operatingSystems[hostPlatform] ?? hostPlatform;
  const hostDevice = `${hostOS} (${hostOSVersion})`;

  /*
   * Read remote device information from BrowserStack
   * capabilities when available.
   *
   * Playwright does not expose remote device capabilities
   * through Page or BrowserContext directly.
   */
  const projectUse = testInfo.project.use as Record<string, unknown>;

  const capabilities =
    projectUse.capabilities && typeof projectUse.capabilities === 'object'
      ? (projectUse.capabilities as Record<string, unknown>)
      : {};

  const browserStackOptions =
    capabilities['bstack:options'] && typeof capabilities['bstack:options'] === 'object'
      ? (capabilities['bstack:options'] as Record<string, unknown>)
      : {};

  const getValue = (value: unknown): string =>
    typeof value === 'string' || typeof value === 'number' ? String(value).trim() : NOT_KNOWN;

  const remoteDevice =
    getValue(browserStackOptions.deviceName) ||
    getValue(capabilities.deviceName) ||
    getValue(process.env.BROWSERSTACK_DEVICE_NAME);

  const remoteOS =
    getValue(browserStackOptions.os) ||
    getValue(capabilities.os) ||
    getValue(process.env.BROWSERSTACK_OS);

  const remoteOSVersion =
    getValue(browserStackOptions.osVersion) ||
    getValue(capabilities.osVersion) ||
    getValue(process.env.BROWSERSTACK_OS_VERSION);

  /*
   * Combine Host OS and Remote Device information.
   *
   * Host only:
   * Windows (10.0.26200)
   *
   * Host + Remote:
   * Windows (10.0.26200) - (iPhone 15 - iOS 17)
   */
  const remoteDetails = [remoteDevice, [remoteOS, remoteOSVersion].filter(Boolean).join(' ')]
    .filter(Boolean)
    .join(' - ');

  const device = remoteDetails ? `${hostDevice} - (${remoteDetails})` : hostDevice;

  const metadata: Record<string, string> = {
    browser: browser?.browserType().name() ?? NOT_KNOWN,

    browserVersion: browser?.version() ?? NOT_KNOWN,

    browserChannel: testInfo.project.use.channel ?? NOT_KNOWN,

    viewport: viewport ? `${viewport.width}x${viewport.height}` : NOT_KNOWN,

    // platform:
    //     hostPlatform,

    // osVersion:
    //     hostOSVersion,

    device,

    headless:
      typeof testInfo.project.use.headless === 'boolean'
        ? String(testInfo.project.use.headless)
        : NOT_KNOWN,
  };

  for (const [type, description] of Object.entries(metadata)) {
    const existing = testInfo.annotations.find((annotation) => annotation.type === type);

    if (!existing) {
      testInfo.annotations.push({
        type,
        description,
      });

      continue;
    }

    // Replace an unknown value if actual metadata becomes available.
    if (
      (!existing.description ||
        existing.description === NOT_KNOWN ||
        existing.description === 'Not Known' ||
        existing.description === 'Not Specified') &&
      description !== NOT_KNOWN
    ) {
      existing.description = description;
    }
  }
}

    const acceptButton = this.page.getByRole('button', { name: 'Accept All', exact: true });
    try {
      await acceptButton.waitFor({ state: 'visible', timeout: 7000 });
    } catch (error) {
      // Only an absent optional banner is expected. Browser/locator failures must surface.
      if (!(error instanceof errors.TimeoutError)) throw error;
      return;
    }
    try {
      await acceptButton.click();
    } catch (error) {
      // Consent can finish asynchronously after the banner was observed. A vanished
      // button is safe only when its blocking overlay has also gone.
      if (
        error instanceof errors.TimeoutError &&
        !(await acceptButton.isVisible()) &&
        !(await this.page.locator('.onetrust-pc-dark-filter:visible').first().isVisible())
      )
        return;
      throw error;
    }

/** Visible destinations remain distinct from hidden templates in these synthetic pages. */
class Destinations {
  constructor(readonly nodes: { visible: () => boolean; click?: () => void; text?: string }[]) {}
  or(other: Destinations) {
    return new Destinations([...this.nodes, ...other.nodes]);
  }
  filter(options: { visible?: boolean }) {
    return new Destinations(
      this.nodes.filter(
        (node) => options.visible === undefined || node.visible() === options.visible
      )
    );
  }
  first() {
    return new Destinations(this.nodes.slice(0, 1));
  }
  async isVisible() {
    return this.nodes[0]?.visible() ?? false;
  }
  async waitFor() {
    if (!(await this.isVisible())) throw new Error('No visible destination');
  }
  async click() {
    if (!(await this.isVisible())) throw new Error('Hidden control');
    this.nodes[0].click?.();
  }
  async innerText() {
    return this.nodes[0]?.text ?? '';
  }
}

  return withLegacyLocators({

    getByRole: (role: string, options: { name: string | RegExp }) =>
      new Destinations([
        {
          visible: () =>

            (options.name instanceof RegExp ? options.name.test(title) : options.name === title),
        },
      ]),
  } as unknown as Page);
}

$oldImport = "import { test, expect } from '@playwright/test';"
$newImport = "import { test, expect } from '@common/baseTest';"

$updated = 0
$skipped = 0
$failed = 0

Get-ChildItem ".\automationLayer" -Recurse -Filter "*.spec.ts" -File |
ForEach-Object {
    try {
        $content = [System.IO.File]::ReadAllText($_.FullName)

        if ($content.Contains($oldImport)) {
            $content = $content.Replace($oldImport, $newImport)

            [System.IO.File]::WriteAllText(
                $_.FullName,
                $content
            )

            $updated++
        }
        else {
            $skipped++
        }
    }
    catch {
        $failed++
        Write-Warning "Failed: $($_.FullName) - $($_.Exception.Message)"
    }
}

Write-Host "Updated: $updated"
Write-Host "Skipped: $skipped"
Write-Host "Failed: $failed"

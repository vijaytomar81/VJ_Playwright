From: 
import { test } from '@playwright/test';

To:
import { test } from '@baseTest';

function prompt { "$(Split-Path (Get-Location) -Leaf)> " }




$target = "from '@playwright/test';"
$replacement = "from '@baseTest';"

$excluded = @(
    'node_modules',
    '.git',
    'test-results',
    'playwright-report',
    'allure-results',
    'allure-report'
)

function Get-SpecFiles {
    param([string]$Path)

    foreach ($file in [System.IO.Directory]::EnumerateFiles($Path, '*.spec.ts')) {
        $file
    }

    foreach ($directory in [System.IO.Directory]::EnumerateDirectories($Path)) {
        if ([System.IO.Path]::GetFileName($directory) -notin $excluded) {
            Get-SpecFiles -Path $directory
        }
    }
}

$files = @(Get-SpecFiles -Path (Get-Location).Path)

$matched = 0
$replaced = 0
$errors = 0
$skipped = 0

foreach ($file in $files) {
    try {
        $longPath = '\\?\' + $file
        $bytes = [System.IO.File]::ReadAllBytes($longPath)

        $hasBom = $bytes.Length -ge 3 -and
            $bytes[0] -eq 0xEF -and
            $bytes[1] -eq 0xBB -and
            $bytes[2] -eq 0xBF

        $utf8 = New-Object System.Text.UTF8Encoding($false, $true)
        $content = $utf8.GetString($bytes)

        if ($content.Contains($target)) {
            $matched++

            $updatedContent = $content.Replace($target, $replacement)

            $encoding = New-Object System.Text.UTF8Encoding($hasBom)
            [System.IO.File]::WriteAllText(
                $longPath,
                $updatedContent,
                $encoding
            )

            $replaced++
        }
        else {
            $skipped++
        }
    }
    catch {
        $errors++
        Write-Warning "Could not process: $file - $($_.Exception.Message)"
    }
}

Write-Host ""
Write-Host "========== Spec Import Replacement Summary =========="
Write-Host "Total .spec.ts Files : $($files.Count)"
Write-Host "Matching Import Files: $matched"
Write-Host "Files Updated        : $replaced"
Write-Host "Files Skipped        : $skipped"
Write-Host "Files With Errors    : $errors"
Write-Host "====================================================="

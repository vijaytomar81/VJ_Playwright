function prompt { "$(Split-Path (Get-Location) -Leaf)> " }



$excluded = @(
    'node_modules',
    '.git',
    'test-results',
    'playwright-report',
    'allure-results',
    'allure-report'
)

function Get-SpecCount {
    param([string]$Path)

    $count = @([System.IO.Directory]::EnumerateFiles($Path, '*.spec.ts')).Count

    foreach ($directory in [System.IO.Directory]::EnumerateDirectories($Path)) {
        if ([System.IO.Path]::GetFileName($directory) -notin $excluded) {
            $count += Get-SpecCount -Path $directory
        }
    }

    return $count
}

$total = Get-SpecCount -Path (Get-Location).Path

Write-Host "Total .spec.ts Files : $total"

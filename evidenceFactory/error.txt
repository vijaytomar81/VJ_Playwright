/**
 * Initializes a unique EvidenceFactory run ID.
 *
 * - Preserves an externally supplied run ID.
 * - Generates a run ID automatically for local execution.
 * - Includes Jenkins BUILD_NUMBER when available.
 *
 * Compatible with Windows, macOS, Linux, CLI and Jenkins.
 */
export function initializeEvidenceRunId(): string {
  const existingRunId = process.env.EVIDENCE_RUN_ID?.trim();

  if (existingRunId) {
    return existingRunId;
  }

  const timestamp = new Date()
    .toISOString()
    .replace(/[-:]/g, '')
    .replace('T', '-')
    .replace(/\.\d{3}Z$/, '');

  const buildNumber = process.env.BUILD_NUMBER?.trim();

  const runId = buildNumber
    ? `RUN-${buildNumber}-${timestamp}`
    : `RUN-${timestamp}`;

  process.env.EVIDENCE_RUN_ID = runId;

  return runId;
}

export async function getInteractionId(page: Page, timeout: number = 20000): Promise<string> {
  await page.waitForURL(/6a[a-z0-9]+$/, { timeout });

  const url = page.url().split('?')[0].replace(/\/$/, '');
  const parts = url.split('/');
  const lastPart = parts.at(-1);

  if (!lastPart?.startsWith('6a')) {
    throw new Error(`Interaction ID not found in URL: ${url}`);
  }

  return lastPart;
}

export async function getPolicyNumber(page: Page): Promise<string | null> {
  const url = page.url().split('?')[0].replace(/\/$/, '');

  const parts = url.split('/');
  const lastPart = parts.at(-1);
  const policyPrefixes = ['BRI'];

  return policyPrefixes.some((prefix) => lastPart?.startsWith(prefix)) ? lastPart : null;
}

export async function getAllCoverPricesForAnnualPayment(page: Page) {
  const fetchTierPrice = async (tierName: string): Promise<string | null> => {
    await page.locator('#selectBRITANNIA_ESSENTIALS').click();
    const tierContainer = page.locator('.tier-premium-display', {
      has: page.locator('.tier-name', { hasText: new RegExp(`^${tierName}$`, 'i') }),
    });

    if ((await tierContainer.count()) === 0) {
      console.warn(`Tier not found: ${tierName}`);
      return null;
    }

    const firstTier = tierContainer.first();
    await firstTier.waitFor({ state: 'visible', timeout: 10000 });

    const priceLocator = firstTier.locator('.price');
    if ((await priceLocator.count()) === 0) {
      console.warn(`Price element not found for tier: ${tierName}`);
      return null;
    }

    await priceLocator.first().waitFor({ state: 'visible', timeout: 10000 });
    const rawText = await priceLocator.first().innerText();
    const cleanedPrice = new RegExp(/£[\d,]+(?:\.\d+)?/).exec(rawText ?? '')?.[0] ?? null;

    return cleanedPrice;
  };

  await fetchAndAttach(
    () => fetchTierPrice('Britannia Essentials'),
    page,
    'BritanniaEssentialsTierAnnualPrice'
  );
  await fetchAndAttach(() => fetchTierPrice('Britannia'), page, 'BritanniaTierAnnualPrice');
  await fetchAndAttach(
    () => fetchTierPrice('Britannia Premier'),
    page,
    'BritanniaPremierTierAnnualPrice'
  );
}

export async function getAllOptionalExtraPrice(page: Page) {
  const fetchOptionalExtraPrice = async (containerId: string, reportKey: string): Promise<void> => {
    const container = page.locator(`#${containerId}.optional-extra`);
    if ((await container.count()) === 0) {
      console.warn(`Optional extra container not found: ${containerId}`);
      await fetchAndAttach(() => null, page, reportKey);
      return;
    }

    const firstContainer = container.first();
    await firstContainer.waitFor({ state: 'attached', timeout: 10000 });

    const includedLocator = firstContainer.locator('.optional-extra-included');
    if ((await includedLocator.count()) > 0) {
      const includedText =
        (await includedLocator.first().innerText())?.trim() ?? 'Included as standard';
      await fetchAndAttach(async () => includedText, page, reportKey);
      return;
    }

    const priceLocator = firstContainer.locator('.optional-extra__price', { hasText: '£' });
    if ((await priceLocator.count()) === 0) {
      console.warn(`Optional extra price not found for ${containerId}`);
      await fetchAndAttach(() => null, page, reportKey);
      return;
    }

    await priceLocator.first().waitFor({ state: 'visible', timeout: 10000 });
    const priceText = (await priceLocator.first().innerText())?.trim();
    if (!priceText) {
      console.warn(`Optional extra price empty for ${containerId}`);
      await fetchAndAttach(() => null, page, reportKey);
      return;
    }

    await fetchAndAttach(async () => priceText, page, reportKey);
  };

  await fetchOptionalExtraPrice('breakdownCover', 'BreakDownCoverPrice');
  await fetchOptionalExtraPrice('motorLegalExpenses', 'MLEPrice');
  await fetchOptionalExtraPrice('courtesyCar', 'GHCPrice');
  await fetchOptionalExtraPrice('optional-extra-protected-ncd', 'NCDPrice');
}

export async function fetchRefundPrice(page: Page): Promise<string | null> {
  const refundLocators: Locator[] = [
    page.locator('p.font-weight-bold:has-text("You will receive") span'),
    page.locator('p.font-weight-semi-bold.pt-2.mb-0', {
      hasText: 'refund',
    }),
  ];

  for (const locator of refundLocators) {
    try {
      await locator.first().waitFor({ state: 'visible', timeout: 5000 });
      const count = await locator.count();
      const text = await locator.nth(count - 1).textContent();

      const regex = /£[\d,.]+/;
      const match = regex.exec(text ?? '');
      if (match) return match[0];
    } catch {}
  }

  const content = await page.content();

  const refundRegex = /£[\d,.]+.*refund|refund.*£[\d,.]+/i;
  const refundMatch = refundRegex.exec(content);

  if (!refundMatch) return null;

  const priceRegex = /£[\d,.]+/;
  const priceMatch = priceRegex.exec(refundMatch[0]);

  return priceMatch ? priceMatch[0] : null;
}

export async function fetchMTCPrice(page: Page): Promise<string> {
  const priceLocator = page.locator('#price');
  await priceLocator.waitFor({ state: 'visible', timeout: 10000 });

  const text = await priceLocator.innerText();
  const regex = /£\s?\d+(\.\d{2})?/;

  }

  return match[0];
}

export async function getPayButtonPrice(page: Page): Promise<string> {
  let payButton = page.locator('button.btn-make-payment');

  if (!(await payButton.count())) {
    payButton = page.getByRole('button', { name: /pay/i });
  }

  if (!(await payButton.count())) {
    payButton = page.locator('button:has-text("£")');
  }

  await payButton.first().waitFor({ state: 'attached' });

  const button = payButton.first();

  let sourceText = await button.getAttribute('aria-label');

  if (!sourceText) {
    sourceText = await button.innerText();
  }

  if (!sourceText) {
    throw new Error('Unable to find price source text on Pay button');
  }

  const regex = /£\s*[\d,.]+/;
  const match = regex.exec(sourceText);

  if (!match) {
    throw new Error(`Unable to extract price from text: ${sourceText}`);
  }

  const price = match[0].replace(/\s+/g, '');

  return price;
}

export async function fetchAndAttach<T extends string | null>(
  fetchFn: (page: Page) => Promise<T>,
  page: Page,
  reportKey: string
): Promise<T> {
  const value = await fetchFn(page);
  if (value) {
    attachValueInTestReport({ [reportKey]: value });
  }
  return value;
}

export async function selectDateFromDatepicker(
  page: Page,
  targetDate: Date,
  options: DatepickerOptions = {}
): Promise<string> {
  const {
    calendarLabel = 'Choose date from calendar',
    confirmButtonName = 'Select date',
    clickNextAfterSelect = true, // default - change to false if DON'T need to click Next button
  } = options;

  const input = page.getByLabel(calendarLabel);
  await input.click();

  const popup = page.locator('.react-datepicker__portal');
  await popup.waitFor({ state: 'visible' });

  const year = targetDate.getFullYear();
  const monthName = targetDate.toLocaleString('default', { month: 'long' });
  const day = targetDate.getDate();

  const monthLocator = popup.locator(
    `.react-datepicker__month[aria-label="Month ${monthName}, ${year}"]`
  );
  await monthLocator.waitFor({ state: 'visible' });

  const dayLocator = monthLocator.locator(
    `.react-datepicker__day:not(.react-datepicker__day--disabled):not(.react-datepicker__day--outside-month)`,
    { hasText: new RegExp(`^${day}$`) }
  );

  await dayLocator.first().click();

  await popup.getByRole('button', { name: confirmButtonName }).click();
  await popup.waitFor({ state: 'detached' });

  const selectedValue = await input.evaluate(
    (el) => (el as HTMLInputElement).value || el.textContent || ''
  );

  if (clickNextAfterSelect) {
    await page.getByRole('button', { name: /^(Next|Get a quote)$/ }).click();
  }

  return selectedValue;
}

export async function selectNextAvailableDate(
  page: Page,
  options: DatepickerOptions = {}
): Promise<string> {
  const { calendarLabel = 'Choose date from calendar', confirmButtonName = 'Select date' } =
    options;

  await page.getByLabel(calendarLabel).click();

  const popup = page.locator('.react-datepicker__portal');
  await popup.waitFor({ state: 'visible' });

  const today = new Date();
  const months = (await page
    .locator('.react-datepicker__month')
    .elementHandles()) as ElementHandle<Element>[];

  const getDateFromDay = async (
    dayEl: ElementHandle<Element>,
    monthEl: ElementHandle<Element>
  ): Promise<Date | null> => {
    const dayText = Number.parseInt((await dayEl.textContent()) ?? '0', 10);
    const monthYearAttr = await monthEl.getAttribute('aria-label');
    if (!monthYearAttr) return null;

    const [monthName, yearStr] = monthYearAttr.replace('Month ', '').split(', ');
    const monthIndex = new Date(`${monthName} 1, ${yearStr}`).getMonth();
    const year = Number.parseInt(yearStr, 10);

    return new Date(year, monthIndex, dayText);
  };

  for (const monthEl of months) {
    const days = await monthEl.$$(
      '.react-datepicker__day[aria-disabled="false"]:not(.disabled):not(.react-datepicker__day--outside-month)'
    );

    for (const dayEl of days) {
      const dateObj = await getDateFromDay(dayEl, monthEl);
      if (!dateObj || dateObj < today) continue;

      await dayEl.click();
      await popup.getByRole('button', { name: confirmButtonName }).click();
      await popup.waitFor({ state: 'detached' });

      const selectedValue = await page
        .getByLabel(calendarLabel)
        .evaluate((el) => (el as HTMLInputElement).value || el.textContent || '');

      await page.getByLabel(calendarLabel).locator('span').first().click();
      await page.getByRole('button', { name: confirmButtonName }).click();

      await page.getByLabel(calendarLabel).locator('span').first().click();
      await page.getByRole('button', { name: confirmButtonName }).click();

      const nextButton = page.getByRole('button', { name: 'Next' });
      const quoteButton = page.getByRole('button', { name: 'Get a quote' });

      if ((await nextButton.count()) > 0 && (await nextButton.isVisible())) {
        await nextButton.click();
      } else if ((await quoteButton.count()) > 0 && (await quoteButton.isVisible())) {
        await quoteButton.click();
      } else {
        throw new Error('Neither Next nor Get a quote button is visible');
      }

      return selectedValue;
    }
  }

  throw new Error('No available date found in the datepicker');
}

export async function fillFutureDate(page: Page, days: number) {
  const date = new Date();
  date.setDate(date.getDate() + days);

  const day = date.getDate().toString().padStart(2, '0');
  const month = (date.getMonth() + 1).toString().padStart(2, '0');
  const year = date.getFullYear().toString();

  await page.locator('#coverStartDateQuestionInput-day').fill(day);
  await page.locator('#coverStartDateQuestionInput-month').fill(month);
  await page.locator('#coverStartDateQuestionInput-year').fill(year);

  await page.getByRole('button', { name: 'Next' }).click();

  const monthNames = [
    'January',
    'February',
    'March',
    'April',
    'May',
    'June',
    'July',
    'August',
    'September',
    'October',
    'November',
    'December',
  ];
  const reportValue = `${Number.parseInt(day)} ${monthNames[date.getMonth()]} ${year}`;

  return reportValue;
}

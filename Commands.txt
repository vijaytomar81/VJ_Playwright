Windows
=========


# Find Microsoft Edge
$edge = "${env:ProgramFiles(x86)}\Microsoft\Edge\Application\msedge.exe"

if (-not (Test-Path $edge)) {
    $edge = "$env:ProgramFiles\Microsoft\Edge\Application\msedge.exe"
}

# Create a dedicated CDP browser profile
$profile = Join-Path $env:TEMP (
    "edge-cdp-" + (Get-Date -Format "yyyyMMdd-HHmmss")
)

# Launch Edge
$edgeProcess = Start-Process `
    -FilePath $edge `
    -ArgumentList @(
        "--remote-debugging-port=9222"
        "--user-data-dir=`"$profile`""
        "--no-first-run"
        "--no-default-browser-check"
    ) `
    -PassThru

# Wait for the CDP endpoint
Start-Sleep -Seconds 2

# Retrieve WebSocket URL
$CDP = (
    Invoke-RestMethod "http://localhost:9222/json/version"
).webSocketDebuggerUrl

Write-Host "CDP URL: $CDP"
Write-Host "Edge PID: $($edgeProcess.Id)"

Mac
=======

PROFILE="/tmp/edge-cdp-$(date +%Y%m%d-%H%M%S)"

open -na "Microsoft Edge" --args \
  --remote-debugging-port=9222 \
  --user-data-dir="$PROFILE"

sleep 2

CDP=$(curl -s http://localhost:9222/json/version | \
  node -pe "JSON.parse(require('fs').readFileSync(0,'utf8')).webSocketDebuggerUrl")

echo "$CDP"

kill browser:

pkill -f "Microsoft Edge.*remote-debugging-port=9222"

=============

Close the CDP browser
---------------------

Stop-Process -Id $edgeProcess.Id

try {
    Invoke-RestMethod "http://localhost:9222/json/version" `
        -TimeoutSec 2 | Out-Null

    Write-Host "CDP still running"
}
catch {
    Write-Host "CDP stopped"
}

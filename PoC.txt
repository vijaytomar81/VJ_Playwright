# ============================================================
# PLAYWRIGHT - CORPORATE PROXY CONFIGURATION
# ============================================================

# 1. Enable Windows corporate certificates
$env:NODE_USE_SYSTEM_CA = "1"

# 2. Configure the detected corporate proxy
$env:HTTPS_PROXY = "http://internet-proxy-a.group.net:81"
$env:HTTP_PROXY  = "http://internet-proxy-a.group.net:81"

# 3. Increase Playwright download timeout
$env:PLAYWRIGHT_DOWNLOAD_CONNECTION_TIMEOUT = "120000"

# 4. Display configuration
Write-Host ""
Write-Host "Corporate proxy configured:"
Write-Host "HTTPS_PROXY = $env:HTTPS_PROXY"
Write-Host "HTTP_PROXY  = $env:HTTP_PROXY"

# 5. Download and install Chromium
npx --no-install playwright install chromium

# 6. Check installation result
if ($LASTEXITCODE -eq 0) {
    Write-Host ""
    Write-Host "SUCCESS: Chromium installed!"
} else {
    Write-Host ""
    Write-Host "ERROR: Chromium installation failed."
}

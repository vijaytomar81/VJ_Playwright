Write-Host "`n=== Corporate Proxy Detection ==="

# Destination used by Playwright
$targetUrl = [Uri]"https://cdn.playwright.dev/builds/cft/156.0.8078.4/win64/chrome-win64.zip"

# Ask Windows/.NET for the proxy selected for this URL
$proxy = [System.Net.WebRequest]::GetSystemWebProxy()

$proxyAddress = $proxy.GetProxy($targetUrl)

Write-Host "Target URL:"
Write-Host $targetUrl

Write-Host "`nProxy selected by Windows:"
Write-Host $proxyAddress

Write-Host "`nDirect connection:"
Write-Host ($proxy.IsBypassed($targetUrl))
